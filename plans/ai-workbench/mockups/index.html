implementation.py
# the agent wrote this; the schema was read from the function
def run(omni, state: str) -> dict:
    issues = omni.call("github.list_issues", state=state)
    local  = omni.call("omnideck.app.storage.list",
                       collection="backlog_items")
    items = [normalize(i) for i in issues] + local.items
    return {"items": items}
